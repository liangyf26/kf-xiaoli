### 1.2 架构特点

- **双层决策架构**：轻量级决策层（快速意图识别） + 重量级生成层（深度推理）
- **3种决策方案可切换**：规则/Jev/Kev，方便对比评估
- **WebSocket实时通信**：支持服务端主动推送（倒计时、状态更新）
- **无状态会话管理**：每个WebSocket连接独立会话，内存存储
- **模块化设计**：各组件职责单一，易于测试和替换

```python
# backend/models.py

from typing import List, Dict, Any, Optional
from datetime import datetime
from pydantic import BaseModel

class Message(BaseModel):
    """单条消息"""
    role: str  # "user" | "assistant" | "system"
    content: str
    timestamp: datetime
    
class ConversationContext(BaseModel):
    """对话上下文"""
    history: List[Message]  # 对话历史
    covered_topics: List[str]  # 已讨论的主题（如"价格_直播线路"）
    user_needs: Dict[str, Any]  # 用户需求画像
    clarification_count: int = 0  # 已澄清次数
    previous_intent: Optional[str] = None  # 上一次意图
    
class WaitingQueue(BaseModel):
    """等待队列"""
    messages: List[str]  # 缓冲的用户消息
    first_message_time: Optional[datetime] = None  # 第一条消息时间
    last_message_time: Optional[datetime] = None  # 最后一条消息时间
    timer_task: Optional[Any] = None  # 定时器任务
    
class SessionState(BaseModel):
    """会话状态"""
    session_id: str
    context: ConversationContext
    waiting_queue: WaitingQueue
    is_first_message: bool = True  # 是否首次对话
    connection: Any  # WebSocket连接对象
```

```python
# backend/connection_manager.py

from fastapi import WebSocket
from typing import Dict
import asyncio

class ConnectionManager:
    """WebSocket连接管理器"""
    
    def __init__(self):
        self.active_sessions: Dict[str, SessionState] = {}
    
    async def connect(self, websocket: WebSocket) -> str:
        """
        接受新连接，创建会话
        
        Returns:
            session_id: 新会话ID
        """
        await websocket.accept()
        session_id = self._generate_session_id()
        
        session = SessionState(
            session_id=session_id,
            context=ConversationContext(history=[], covered_topics=[], user_needs={}),
            waiting_queue=WaitingQueue(messages=[]),
            connection=websocket
        )
        
        self.active_sessions[session_id] = session
        return session_id
    
    def disconnect(self, session_id: str):
        """断开连接，清理会话"""
        if session_id in self.active_sessions:
            session = self.active_sessions[session_id]
            # 取消等待定时器
            if session.waiting_queue.timer_task:
                session.waiting_queue.timer_task.cancel()
            del self.active_sessions[session_id]
    
    async def send_message(self, session_id: str, message: dict):
        """发送消息给客户端"""
        session = self.active_sessions.get(session_id)
        if session:
            await session.connection.send_json(message)
    
    async def broadcast_status(self, session_id: str, status: str, data: dict = None):
        """推送状态更新"""
        await self.send_message(session_id, {
            "type": "status",
            "status": status,
            "data": data or {}
        })
```

### 2.2 等待汇总层 (Wait Aggregator)

#### 职责
- 实现滑动窗口机制
- 缓冲用户连续消息
- 决定何时触发处理

#### 核心逻辑

```python
# backend/wait_aggregator.py

import asyncio
from datetime import datetime, timedelta
from typing import Callable
from config import settings

class WaitAggregator:
    """等待汇总处理器"""
    
    def __init__(
        self, 
        slide_seconds: int = 15,
        max_seconds: int = 30
    ):
        self.slide_seconds = slide_seconds  # 滑动窗口
        self.max_seconds = max_seconds  # 最长等待
    
    async def add_message(
        self,
        session: SessionState,
        message: str,
        on_timeout: Callable
    ):
        """
        添加用户消息到等待队列
        
        Args:
            session: 会话状态
            message: 用户消息
            on_timeout: 超时回调函数
        """
        queue = session.waiting_queue
        now = datetime.now()
        
        # 添加消息到缓冲区
        queue.messages.append(message)
        
        # 记录时间
        if queue.first_message_time is None:
            queue.first_message_time = now
        queue.last_message_time = now
        
        # 取消旧的定时器
        if queue.timer_task:
            queue.timer_task.cancel()
        
        # 计算剩余等待时间
        elapsed = (now - queue.first_message_time).total_seconds()
        remaining_max = self.max_seconds - elapsed
        
        if remaining_max <= 0:
            # 已达最长等待时间，立即处理
            await self._trigger_processing(session, on_timeout)
        else:
            # 启动新的滑动窗口定时器
            wait_time = min(self.slide_seconds, remaining_max)
            queue.timer_task = asyncio.create_task(
                self._wait_and_trigger(session, wait_time, on_timeout)
            )
    
    async def _wait_and_trigger(
        self,
        session: SessionState,
        wait_seconds: float,
        on_timeout: Callable
    ):
        """等待指定时间后触发处理"""
        try:
            # 等待期间推送倒计时
            for remaining in range(int(wait_seconds), 0, -1):
                await session.connection.send_json({
                    "type": "waiting",
                    "remaining_seconds": remaining
                })
                await asyncio.sleep(1)
            
            # 倒计时结束，触发处理
            await self._trigger_processing(session, on_timeout)
            
        except asyncio.CancelledError:
            # 定时器被取消（新消息到达）
            pass
    
    async def _trigger_processing(
        self,
        session: SessionState,
        on_timeout: Callable
    ):
        """触发消息处理"""
        queue = session.waiting_queue
        
        if queue.messages:
            # 汇总消息
            aggregated_message = " ".join(queue.messages)
            
            # 清空队列
            queue.messages = []
            queue.first_message_time = None
            queue.last_message_time = None
            queue.timer_task = None
            
            # 调用处理回调
            await on_timeout(session, aggregated_message)
```

```python
from abc import ABC, abstractmethod
from typing import Dict, Any
from pydantic import BaseModel

class DecisionResult(BaseModel):
    """决策结果（统一格式）"""
    intent: str  # 主要意图
    intent_confidence: float  # 置信度 0-1
    needs_clarification: bool  # 是否需要澄清
    clarification_reason: str = ""  # 澄清原因
    technical_complexity: int  # 技术复杂度 0-100
    user_emotion: str  # 用户情绪
    escalate_to_human: bool  # 是否转人工
    raw_response: Dict[str, Any] = {}  # 原始响应（调试用）
    latency_ms: int  # 推理耗时
    engine: str  # 使用的引擎名称

class DecisionEngine(ABC):
    """决策引擎抽象基类"""
    
    @abstractmethod
    async def decide(
        self,
        message: str,
        context: Dict[str, Any]
    ) -> DecisionResult:
        """
        执行决策
        
        Args:
            message: 用户消息
            context: 上下文信息
            
        Returns:
            DecisionResult: 统一格式的决策结果
        """
        pass
```

#### 2.3.2 方案1: 规则引擎

```python
# backend/decision_layer/rule_engine.py

import re
from datetime import datetime

class RuleBasedEngine(DecisionEngine):
    """基于规则的决策引擎"""
    
    def __init__(self):
        # 意图识别规则（关键词）
        self.intent_keywords = {
            "price_inquiry": ["多少钱", "价格", "费用", "收费", "元", "钱"],
            "technical_support": ["连不上", "不上", "登不上", "错误", "失败", "问题", "故障"],
            "usage_guide": ["怎么", "如何", "怎样", "咋", "咋整", "用法", "使用"],
            "product_comparison": ["能不能", "可以", "支持", "有没有", "区别", "对比"],
            "troubleshooting": ["卡", "慢", "掉线", "断开", "不稳定"],
            "purchase_process": ["购买", "买", "订购", "下单"],
        }
        
        # 澄清触发规则
        self.clarification_triggers = [
            (r"^.{1,5}$", "消息过短"),  # 5字以内
            (r"^(多少钱|咋整|怎么办|啥|哪个)$", "单一疑问词"),
        ]
        
        # 情绪识别规则
        self.emotion_keywords = {
            "negative": ["不行", "垃圾", "差", "坑", "骗", "退款", "投诉"],
            "urgent": ["紧急", "赶紧", "快", "马上", "急"],
            "positive": ["好", "不错", "谢谢", "感谢"],
        }
    
    async def decide(
        self,
        message: str,
        context: Dict[str, Any]
    ) -> DecisionResult:
        start_time = datetime.now()
        
        # 1. 意图识别
        intent, confidence = self._match_intent(message)
        
        # 2. 判断是否需要澄清
        needs_clarification = self._check_clarification(
            message, 
            intent, 
            confidence,
            context
        )
        
        # 3. 技术复杂度评估
        complexity = self._estimate_complexity(message, intent)
        
        # 4. 情绪识别
        emotion = self._detect_emotion(message)
        
        # 5. 判断是否转人工
        escalate = self._should_escalate(emotion, complexity, context)
        
        latency = int((datetime.now() - start_time).total_seconds() * 1000)
        
        return DecisionResult(
            intent=intent,
            intent_confidence=confidence,
            needs_clarification=needs_clarification,
            clarification_reason="上下文不足或问题模糊" if needs_clarification else "",
            technical_complexity=complexity,
            user_emotion=emotion,
            escalate_to_human=escalate,
            latency_ms=latency,
            engine="rule"
        )
    
    def _match_intent(self, message: str) -> tuple[str, float]:
        """匹配意图（关键词）"""
        scores = {}
        for intent, keywords in self.intent_keywords.items():
            score = sum(1 for kw in keywords if kw in message)
            if score > 0:
                scores[intent] = score
        
        if not scores:
            return "unclear", 0.3
        
        # 返回最高分意图
        best_intent = max(scores, key=scores.get)
        max_score = scores[best_intent]
        confidence = min(0.95, 0.5 + max_score * 0.2)  # 规则引擎置信度上限0.95
        
        return best_intent, confidence
    
    def _check_clarification(
        self, 
        message: str, 
        intent: str, 
        confidence: float,
        context: Dict[str, Any]
    ) -> bool:
        """判断是否需要澄清"""
        # 规则1: 已澄清2次以上，不再澄清
        if context.get("clarification_count", 0) >= 2:
            return False
        
        # 规则2: 置信度低
        if confidence < 0.6:
            return True
        
        # 规则3: 短问题且无上下文
        if len(message) <= 5:
            previous_intent = context.get("previous_intent")
            if not previous_intent or intent != previous_intent:
                return True
        
        # 规则4: 触发澄清规则
        for pattern, reason in self.clarification_triggers:
            if re.match(pattern, message):
                return True
        
        return False
    
    def _estimate_complexity(self, message: str, intent: str) -> int:
        """估算技术复杂度"""
        # 简单规则
        if intent in ["price_inquiry", "product_comparison"]:
            return 20
        elif intent in ["usage_guide"]:
            return 40
        elif intent in ["technical_support", "troubleshooting"]:
            return 60
        else:
            return 50
    
    def _detect_emotion(self, message: str) -> str:
        """检测用户情绪"""
        for emotion, keywords in self.emotion_keywords.items():
            if any(kw in message for kw in keywords):
                return emotion
        return "neutral"
    
    def _should_escalate(
        self, 
        emotion: str, 
        complexity: int,
        context: Dict[str, Any]
    ) -> bool:
        """判断是否转人工"""
        # 规则1: 负面情绪 + 高复杂度
        if emotion == "negative" and complexity > 50:
            return True
        
        # 规则2: 明确要求人工
        # (这里简化，实际可在关键词匹配时标记)
        
        # 规则3: 超出知识库范围（复杂度>80）
        if complexity > 80:
            return True
        
        return False
```

```python
# backend/decision_layer/jev_client.py

import httpx
from datetime import datetime
from config import settings

class JevEngine(DecisionEngine):
    """Jev API决策引擎"""
    
    def __init__(self, api_key: str):
        self.api_key = api_key
        self.base_url = "https://api.typesafe.com/v1"  # 假设URL
        
        # 问题定义（与PRD中的yaml对应）
        self.questions = {
            "intent": {
                "kind": "choice",
                "options": [
                    "price_inquiry",
                    "product_comparison",
                    "technical_support",
                    "usage_guide",
                    "troubleshooting",
                    "purchase_process",
                    "unclear"
                ]
            },
            "needs_clarification": {
                "kind": "noul"
            },
            "intent_confidence": {
                "kind": "score",
                "min": 0,
                "max": 100
            },
            "user_emotion": {
                "kind": "choice",
                "options": ["neutral", "positive", "negative", "urgent"]
            },
            "technical_complexity": {
                "kind": "score",
                "min": 0,
                "max": 100
            },
            "escalate_to_human": {
                "kind": "noul"
            }
        }
    
    async def decide(
        self,
        message: str,
        context: Dict[str, Any]
    ) -> DecisionResult:
        start_time = datetime.now()
        
        # 构建state
        state = {
            "message": message,
            "conversation_history": context.get("history", [])[-10:],  # 最近10条
            "covered_topics": context.get("covered_topics", []),
            "previous_intent": context.get("previous_intent"),
            "clarification_count": context.get("clarification_count", 0)
        }
        
        # 调用Jev API
        try:
            async with httpx.AsyncClient(timeout=5.0) as client:
                response = await client.post(
                    f"{self.base_url}/evaluate",
                    json={"state": state, "questions": self.questions},
                    headers={"Authorization": f"Bearer {self.api_key}"}
                )
                response.raise_for_status()
                data = response.json()
                
            latency = int((datetime.now() - start_time).total_seconds() * 1000)
            
            # 解析响应
            answers = data.get("answers", {})
            
            return DecisionResult(
                intent=answers["intent"]["choice"],
                intent_confidence=answers["intent_confidence"]["score"] / 100.0,
                needs_clarification=answers["needs_clarification"]["noul"],
                clarification_reason="Jev判断需要澄清" if answers["needs_clarification"]["noul"] else "",
                technical_complexity=answers["technical_complexity"]["score"],
                user_emotion=answers["user_emotion"]["choice"],
                escalate_to_human=answers["escalate_to_human"]["noul"],
                raw_response=data,
                latency_ms=latency,
                engine="jev"
            )
            
        except Exception as e:
            # API调用失败，回退到低置信度
            return DecisionResult(
                intent="unclear",
                intent_confidence=0.0,
                needs_clarification=True,
                clarification_reason=f"Jev API错误: {str(e)}",
                technical_complexity=50,
                user_emotion="neutral",
                escalate_to_human=False,
                latency_ms=int((datetime.now() - start_time).total_seconds() * 1000),
                engine="jev_failed"
            )
```

```python
# backend/decision_layer/kev_client.py

from transformers import AutoModelForCausalLM, AutoTokenizer
import torch
from datetime import datetime
import json

class KevEngine(DecisionEngine):
    """Kev本地模型决策引擎"""
    
    def __init__(self, model_path: str = "tt-hous/kev-0.5b", device: str = "cpu"):
        self.device = device
        self.model = AutoModelForCausalLM.from_pretrained(
            model_path,
            torch_dtype=torch.float16 if device != "cpu" else torch.float32,
            device_map="auto" if device != "cpu" else None
        )
        self.tokenizer = AutoTokenizer.from_pretrained(model_path)
        
        if device == "cpu":
            self.model = self.model.to("cpu")
        
        self.model.eval()
    
    async def decide(
        self,
        message: str,
        context: Dict[str, Any]
    ) -> DecisionResult:
        start_time = datetime.now()
        
        # 构建Kev输入格式
        # (注: 实际格式需参考Kev官方文档，这里简化)
        prompt = self._build_kev_prompt(message, context)
        
        # 推理
        with torch.no_grad():
            inputs = self.tokenizer(prompt, return_tensors="pt")
            if self.device != "cpu":
                inputs = {k: v.to(self.device) for k, v in inputs.items()}
            
            outputs = self.model.generate(
                **inputs,
                max_new_tokens=256,
                temperature=0.3,
                do_sample=False
            )
            
            response_text = self.tokenizer.decode(outputs[0], skip_special_tokens=True)
        
        latency = int((datetime.now() - start_time).total_seconds() * 1000)
        
        # 解析Kev输出（假设返回JSON）
        try:
            result = json.loads(response_text.split("```json")[-1].split("```")[0])
            
            return DecisionResult(
                intent=result.get("intent", "unclear"),
                intent_confidence=result.get("intent_confidence", 0.5),
                needs_clarification=result.get("needs_clarification", False),
                clarification_reason=result.get("clarification_reason", ""),
                technical_complexity=result.get("technical_complexity", 50),
                user_emotion=result.get("user_emotion", "neutral"),
                escalate_to_human=result.get("escalate_to_human", False),
                raw_response=result,
                latency_ms=latency,
                engine="kev"
            )
        except:
            # 解析失败，返回默认值
            return DecisionResult(
                intent="unclear",
                intent_confidence=0.3,
                needs_clarification=True,
                clarification_reason="Kev输出解析失败",
                technical_complexity=50,
                user_emotion="neutral",
                escalate_to_human=False,
                latency_ms=latency,
                engine="kev_failed"
            )
    
    def _build_kev_prompt(self, message: str, context: Dict[str, Any]) -> str:
        """构建Kev prompt（简化版）"""
```

```python
# backend/orchestrator/router.py

from enum import Enum
from typing import Optional
from backend.decision_layer.base import DecisionResult

class ProcessingPath(Enum):
    """处理路径"""
    CLARIFICATION = "clarification"
    FAQ_MATCH = "faq_match"
    LLM_GENERATION = "llm_generation"
    HUMAN_ESCALATION = "human_escalation"

class Router:
    """路由决策器"""
    
    def determine_path(
        self,
        decision: DecisionResult,
        context: Dict[str, Any]
    ) -> ProcessingPath:
        """
        根据决策结果确定处理路径
        
        优先级:
        1. 转人工 (最高优先级)
        2. 需要澄清
        3. 简单问题 (FAQ直接匹配)
        4. 复杂问题 (LLM生成)
        5. 超出能力 (转人工)
        """
        
        # P1: 转人工
        if decision.escalate_to_human:
            return ProcessingPath.HUMAN_ESCALATION
        
        # P2: 需要澄清
        if decision.needs_clarification:
            # 智能限制: 已澄清2次以上不再澄清
            clarification_count = context.get("clarification_count", 0)
            if clarification_count >= 2:
                # 置信度太低就转人工，否则尝试回答
                if decision.intent_confidence < 0.5:
                    return ProcessingPath.HUMAN_ESCALATION
                # 否则尝试LLM生成
            else:
                return ProcessingPath.CLARIFICATION
        
        # P3: 简单问题 (低复杂度 + 高置信度)
        if (decision.technical_complexity < 30 and 
            decision.intent_confidence > 0.8):
            return ProcessingPath.FAQ_MATCH
        
        # P4: 中等复杂度，用LLM
        if decision.technical_complexity < 80:
            return ProcessingPath.LLM_GENERATION
        
        # P5: 超出能力范围
        return ProcessingPath.HUMAN_ESCALATION


class Orchestrator:
    """核心编排器"""
    
    def __init__(
        self,
        decision_engine,
        llm_client,
        knowledge_base,
        router: Router
    ):
        self.decision_engine = decision_engine
        self.llm = llm_client
        self.kb = knowledge_base
        self.router = router
    
    async def process_message(
        self,
        message: str,
        session: SessionState
    ) -> Dict[str, Any]:
        """
        处理用户消息的主流程
        
        Returns:
            {
                "answer": "回复内容",
                "sources": ["问题3"],
                "path": "llm_generation",
                "decision": {...},
                "need_clarification": false
            }
        """
        context = session.context
        
        # Step 1: 决策层判断
        decision = await self.decision_engine.decide(
            message=message,
            context={
                "history": [m.dict() for m in context.history],
                "covered_topics": context.covered_topics,
                "clarification_count": context.clarification_count,
                "previous_intent": context.previous_intent
            }
        )
        
        # Step 2: 路由决策
        path = self.router.determine_path(decision, context.dict())
        
        # Step 3: 根据路径处理
        if path == ProcessingPath.CLARIFICATION:
            result = await self._handle_clarification(message, decision, context)
            context.clarification_count += 1
            
        elif path == ProcessingPath.FAQ_MATCH:
            result = await self._handle_faq(decision, context)
            
        elif path == ProcessingPath.LLM_GENERATION:
            result = await self._handle_llm(message, decision, context)
            context.clarification_count = 0  # 重置澄清计数
            
        elif path == ProcessingPath.HUMAN_ESCALATION:
            result = self._handle_escalation(decision, context)
        
        # Step 4: 更新上下文
        context.previous_intent = decision.intent
        if result.get("sources"):
            context.covered_topics.extend([
                f"{decision.intent}_{src}" for src in result["sources"]
            ])
        
        # 添加到历史
        context.history.append(Message(role="user", content=message, timestamp=datetime.now()))
        context.history.append(Message(role="assistant", content=result["answer"], timestamp=datetime.now()))
        
        # 保留最近N轮
        max_turns = settings.CONTEXT_TURNS * 2  # 用户+助手各N条
        if len(context.history) > max_turns:
            context.history = context.history[-max_turns:]
        
        result["decision"] = decision.dict()
        result["path"] = path.value
        
        return result
    
    async def _handle_clarification(self, message, decision, context):
        """生成澄清问题"""
        # 简化实现：基于意图生成澄清
        templates = {
            "price_inquiry": "您是想咨询哪种线路的价格？\n1. IDC线路（120元/月）\n2. ISP家庭IP线路（180元/月）\n3. 直播优化线路（260元/月）\n4. 拼车共享（50元/月/人）",
            "technical_support": "请问您遇到的是什么问题？\n1. 无法连接/登录\n2. 速度慢/卡顿\n3. 设备配置问题\n4. 其他",
            "unclear": "抱歉，我没太理解您的问题。请问您是想：\n1. 咨询价格和套餐\n2. 了解产品功能和使用方法\n3. 解决技术问题\n4. 其他咨询"
        }
        
        clarification = templates.get(decision.intent, templates["unclear"])
        
        return {
            "answer": clarification,
            "sources": [],
            "need_clarification": True
        }
    
    async def _handle_faq(self, decision, context):
        """FAQ快速匹配"""
        # 根据意图获取相关知识库内容
        relevant_qa = self.kb.get_by_intent(decision.intent, limit=1)
        
        if relevant_qa:
            return {
                "answer": relevant_qa["content"],
                "sources": [f"问题{relevant_qa['number']}"],
                "need_clarification": False
            }
        else:
            # 回退到LLM
            return await self._handle_llm("", decision, context)
    
    async def _handle_llm(self, message, decision, context):
        """LLM生成回复"""
        # 构建prompt
        prompt = self._build_llm_prompt(message, decision, context)
        
        # 调用LLM
        response = await self.llm.generate(prompt)
        
        # 解析结构化输出
        parsed = self._parse_llm_response(response)
        
        return {
            "answer": parsed["answer"],
            "sources": parsed.get("sources", []),
            "need_clarification": False
        }
    
    def _handle_escalation(self, decision, context):
        """转人工"""
        return {
            "answer": "您的问题比较复杂，已为您转接人工客服，请稍候...",
            "sources": [],
            "need_clarification": False,
            "escalated": True
        }
    
    def _build_llm_prompt(self, message, decision, context):
        """构建LLM prompt（下一节详述）"""

        """解析LLM JSON输出"""
        try:
            return json.loads(response)
        except:
            # 解析失败，尝试提取
            return {
                "answer": response,
                "sources": []
```

```python
# backend/knowledge.py

import re
from typing import List, Dict, Optional
from pathlib import Path

class QAPair:
    """问答对"""
    def __init__(self, number: str, title: str, content: str, category: str):
        self.number = number
        self.title = title
        self.content = content
        self.category = category
    
    def to_dict(self) -> Dict:
        return {
            "number": self.number,
            "title": self.title,
            "content": self.content,
            "category": self.category
        }

class KnowledgeBase:
    """知识库管理器"""
    
    # 意图到类别的映射
    INTENT_CATEGORY_MAP = {
        "price_inquiry": ["price", "product"],
        "technical_support": ["technical", "troubleshooting"],
        "usage_guide": ["usage", "installation"],
        "product_comparison": ["product", "features"],
        "troubleshooting": ["troubleshooting", "technical"],
        "purchase_process": ["purchase", "general"]
    }
    
    def __init__(self, file_path: str):
        self.file_path = Path(file_path)
        self.qa_pairs: List[QAPair] = []
        self.category_index: Dict[str, List[QAPair]] = {}
        
        self._load()
    
    def _load(self):
        """加载并解析知识库"""
        if not self.file_path.exists():
            raise FileNotFoundError(f"知识库文件不存在: {self.file_path}")
        
        content = self.file_path.read_text(encoding='utf-8')
        
        if not content.strip():
            raise ValueError("知识库文件为空")
        
        # 解析知识库
        try:
            self._parse_markdown(content)
            print(f"✓ 已加载 {len(self.qa_pairs)} 个问答对")
        except Exception as e:
            print(f"⚠ 知识库解析失败: {e}，回退到纯文本模式")
            self._fallback_mode(content)
    
    def _parse_markdown(self, content: str):
        """解析Markdown格式知识库"""
        lines = content.split('\n')
        pattern = r'^(\d+)\.\s+(.+?)$'
        
        current_qa = None
        
        for line in lines:
            match = re.match(pattern, line)
            if match:
                # 保存上一个QA
                if current_qa:
                    self._add_qa(current_qa)
                
                # 开始新的QA
                number = match.group(1)
                title = match.group(2).strip()
                current_qa = {
                    "number": number,
                    "title": title,
                    "content_lines": []
                }
            elif current_qa and line.strip():
                # 内容行
                current_qa["content_lines"].append(line)
        
        # 最后一个QA
        if current_qa:
            self._add_qa(current_qa)
    
    def _add_qa(self, qa_dict: Dict):
        """添加问答对"""
        content = "\n".join(qa_dict["content_lines"]).strip()
        if not content:
            return
        
        # 分类
        category = self._categorize(qa_dict["title"], content)
        
        qa = QAPair(
            number=qa_dict["number"],
            title=qa_dict["title"],
            content=content,
            category=category
        )
        
        self.qa_pairs.append(qa)
        
        # 建立索引
        if category not in self.category_index:
            self.category_index[category] = []
        self.category_index[category].append(qa)
    
    def _categorize(self, title: str, content: str) -> str:
        """自动分类（基于规则）"""
        text = title + " " + content
        
        # 规则匹配
        if any(kw in text for kw in ["价格", "多少钱", "元", "费用", "收费"]):
            return "price"
        elif any(kw in text for kw in ["怎么", "如何", "安装", "使用", "教程", "客户端"]):
            return "usage"
        elif any(kw in text for kw in ["连不上", "登不上", "错误", "问题", "卡", "慢"]):
            return "troubleshooting"
        elif any(kw in text for kw in ["什么是", "能不能", "可以", "支持", "有没有"]):
            return "product"
        elif any(kw in text for kw in ["购买", "下单", "支付", "发票"]):
            return "purchase"
        elif any(kw in text for kw in ["设备", "路由", "硬件", "配置"]):
            return "technical"
        else:
            return "general"
    
    def _fallback_mode(self, content: str):
        """回退模式（不解析结构，整体当作一个QA）"""
        self.qa_pairs.append(QAPair(
            number="1",
            title="全部知识",
            content=content,
            category="general"
        ))
    
    def get_by_intent(self, intent: str, limit: Optional[int] = None) -> str:
        """根据意图获取知识库内容"""
        categories = self.INTENT_CATEGORY_MAP.get(intent, ["general"])
        
        selected_qa = []
        for cat in categories:
            if cat in self.category_index:
                selected_qa.extend(self.category_index[cat])
        
        if not selected_qa:
            # 回退到全部
            selected_qa = self.qa_pairs
        
        if limit:
            selected_qa = selected_qa[:limit]
        
        # 格式化输出
        lines = []
        for qa in selected_qa:
            lines.append(f"【问题{qa.number}】{qa.title}")
            lines.append(qa.content)
            lines.append("")
        
        return "\n".join(lines)
    
    def get_all(self) -> str:
        """获取全部知识库"""
        return self.get_by_intent("general")
    
    def search_by_number(self, number: str) -> Optional[QAPair]:
        """根据编号查找"""
        for qa in self.qa_pairs:
            if qa.number == number:
                return qa
        return None
```

```python
async def process_with_timeout(decision_engine, message, context, timeout=5.0):
    """带超时的决策调用"""
    try:
        return await asyncio.wait_for(
```

## 七、关键技术点

### 7.1 滑动窗口实现

使用`asyncio.create_task`创建定时器任务，每次新消息到达时取消旧任务并创建新任务，实现滑动窗口效果。通过`first_message_time`追踪最长等待时间。

### 7.2 决策层可插拔设计

3种决策引擎实现统一接口`DecisionEngine`，通过配置文件`DECISION_ENGINE`参数切换，便于对比测试。

### 7.3 Few-shot Prompting

在system prompt中嵌入3-10个真实对话示例，教会模型：
- 如何处理短问题和模糊问题
- 如何利用上下文
- 如何标注答案来源
- 如何拒绝回答知识库外问题

### 7.4 结构化输出

强制LLM返回JSON格式：`{"answer": "...", "sources": [...]}`，便于解析和展示。使用正则或JSON parser提取，容错处理格式错误。

### 7.5 上下文管理

保留最近N轮对话（配置CONTEXT_TURNS），超过限制自动截断。维护`covered_topics`列表避免重复回答。

### 8.2 性能指标

```python
# 性能追踪
class PerformanceTracker:
    def __init__(self):
        self.metrics = {
            "total_requests": 0,
            "avg_latency_ms": 0,
            "intent_distribution": {},
            "path_distribution": {},
            "engine_latency": {}
        }
    
    def record(self, decision: DecisionResult, path: str, total_latency: int):
        self.metrics["total_requests"] += 1
        self.metrics["intent_distribution"][decision.intent] = \
            self.metrics["intent_distribution"].get(decision.intent, 0) + 1
        self.metrics["path_distribution"][path] = \
            self.metrics["path_distribution"].get(path, 0) + 1
        
        # 更新平均延迟
        n = self.metrics["total_requests"]
        old_avg = self.metrics["avg_latency_ms"]
        self.metrics["avg_latency_ms"] = (old_avg * (n - 1) + total_latency) / n
        
        # 记录引擎延迟
        engine = decision.engine
        if engine not in self.metrics["engine_latency"]:
            self.metrics["engine_latency"][engine] = []
        self.metrics["engine_latency"][engine].append(decision.latency_ms)
    
    def get_summary(self) -> dict:
        summary = self.metrics.copy()
        # 计算各引擎平均延迟
        for engine, latencies in summary["engine_latency"].items():
            summary["engine_latency"][engine] = sum(latencies) / len(latencies)
        return summary
```

## 十一、风险和应对

### 11.1 技术风险

| 风险 | 影响 | 应对措施 |
|------|------|---------|
| Jev API中文能力不足 | 意图识别准确率低 | 提前测试中文案例，不理想则主推Kev方案 |
| Kev-0.5B推理速度慢 | 响应时间不达标 | 使用GPU加速，或升级到云端推理 |
| Qwen27B推理超时 | 用户体验差 | 设置60秒超时，超时提示并转人工 |
| WebSocket连接不稳定 | 消息丢失 | 实现自动重连，本地消息队列缓冲 |

### 11.2 业务风险

| 风险 | 影响 | 应对措施 |
|------|------|---------|
| 知识库更新频繁 | 重启服务影响可用性 | 后续实现热更新机制 |
| 真实案例不足 | Few-shot示例不准确 | 先用假想示例，收集真实案例后迭代 |
| 准确率不达标 | 无法投入生产 | 降级到纯FAQ模式，或人工审核后回复 |
