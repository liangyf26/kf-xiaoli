### 逐题对比复算命令

```
cd /home/lyf/project/kf-xiaoli && .venv/bin/python - <<'EOF'
import json

def load(p):
    with open(p, encoding="utf-8") as fh:
        return json.load(fh)

def correct(r):
    if r["expected_action"] == "answer":
        return bool(r["answer_pass"])
    if r["expected_action"] == "handoff":
        return bool(r["handoff_pass"])
    return None

new = load("tests/results/accuracy_eval_rule_20261027-024016.json")
old = load("tests/results/accuracy_eval_rule_20261026-194542.json")
nm = {(r["batch"], r["question"]): correct(r) for r in new["detailed_results"] if correct(r) is not None}
om = {(r["batch"], r["question"]): correct(r) for r in old["detailed_results"]
      if (r.get("batch") or "").startswith("B") and correct(r) is not None}
common = sorted(set(nm) & set(om))
worse = [k for k in common if om[k] and not nm[k]]
better = [k for k in common if not om[k] and nm[k]]
print(f"旧060 {sum(om[k] for k in common)}/{len(common)}  新 {sum(nm[k] for k in common)}/{len(common)}  "
      f"翻转{len(worse)+len(better)}（变差{len(worse)} 变好{len(better)}）")
EOF
```

实测输出：`旧060 37/70  新 34/70  翻转7（变差5 变好2）`。
